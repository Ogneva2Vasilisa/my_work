## Решение

Сначала файл был разархивирован с помощью UPX для статического анализа. 
Обход дебага осуществлялся заменой проверок на конструкцию ```nop``` . В итоге, получилось запустить программу в IDA:

Далее, пройдясь по веткам проверки пароля, можно достать зашифрованный пароль. Теперь можно написать небольшую программу для расшифровки пароля или использовать плагин **Ponce**.

Небольшая программа:

```C

    unsigned char a6g1ecpknCn18gE [50] = "6G1ECPKN>CN18G;ED8RQD:=7:@;7MTEDQAK2SGM5:IHNQ>AEM2";

    unsigned char a22233301232120 [50] = "22233301232120101011320313033011000313301133003212";

    unsigned char v7[38] = { 0x53, 0x8D, 0x51, 0x65, 0x72, 0x52, 0x98, 0x18, 0xA4, 0x65, 0xA7, 0x6E, 0x61, 0x62, 0x9D, 0x64, 0x37, 0x36, 0x72, 0x14, 0x08, 0x44, 0x73, 0x6D, 0x37, 0x5D, 0x64, 0x2E, 0x31, 0x07, 0x63, 0x34, 0x1A, 0x1B, 0x18, 0x17, 0x38, 0x00};

    for (int j = 0; j < 50; ++j )
      {
        char v6 = a6g1ecpknCn18gE[j] - 48;
        if ( a22233301232120[j] == 49 )
        {
          v7[v6] ^= 47u;
        }
        else if ( a22233301232120[j] == 48 )
        {
            ++v7[v6];
        }
        else if ( a22233301232120[j] == 50 )
        {
          v7[v6] -= 51;
        }
        else if ( a22233301232120[j] == 51 )
        {
          v7[v6] = v7[v6] * 2;
        }
      }

  for (int i=0; i<38;i++)
  printf("%c", v7[i]);
```
