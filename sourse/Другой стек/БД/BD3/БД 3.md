## Построение цепочки

### Вариант 1 (самый логичный и типичный для анализа сетевых событий)

Цепочка «Анализ дневного трафика → Поиск перегруженных дней → Определение источников трафика в аномальные дни»
#### Шаг 1 — MR-1: трафик по дням
`events → mr_traffic_per_day`
#### Шаг 2 — MR-2: выявить аномальные дни (дни, где трафик > среднего значения)
`mr_traffic_per_day → mr_anomalous_days`
(Здесь MR будет брать key = day, value = total_bytes и фильтровать.)
#### Шаг 3 — MR-3: по аномальным дням собрать трафик по IP
`events → mr_top_ips_in_anomaly`,  
но с фильтрацией только для дней из `mr_anomalous_days`.

Итого: цепочка превращается в поток аналитики:
> дневная агрегация → фильтр → детализация по IP

 **MR-2 — Выявить аномальные дни**
Подход:
- вычисляем средний трафик
- MR выбирает дни, где `bytes > avg`
```
// ===== MR-2: ВЫЯВЛЕНИЕ АНОМАЛЬНЫХ ДНЕЙ =====

// 1. Считаем среднее значение трафика по дням
var avgTraffic = db.mr_traffic_per_day.aggregate([
    { $group: { _id: null, avgBytes: { $avg: "$value" } } }
]).toArray()[0].avgBytes;

print("Среднее значение трафика:", avgTraffic);

// 2. MAP: каждый день -> трафик, но фильтруем только аномальные
var map_anomalous_days = function() {
    if (this.value > avg_traffic) {
        emit(this._id, this.value);
    }
};

// 3. REDUCE: просто возвращает значение (на случай группировки)
var reduce_anomalous_days = function(key, values) {
    return Array.sum(values);
};

// 4. Запуск MapReduce
db.mr_traffic_per_day.mapReduce(
    map_anomalous_days,
    reduce_anomalous_days,
    {
        out: "mr_anomalous_days",
        scope: { avg_traffic: avgTraffic }
    }
);

print("MR-2 завершён: mr_anomalous_days создана");
```

**MR-3 — Для аномальных дней посчитать трафик по Source IP**
Тут:
- MR читает исходную коллекцию 
- фильтрует по датам, которые попали в `mr_anomalous_days`
- агрегирует по `source_ip`

```
// ===== MR-3: ТРАФИК ПО SOURCE IP ДЛЯ АНОМАЛЬНЫХ ДНЕЙ =====

// 0. Получаем список аномальных дат в удобном формате
var anomalousDates = db.mr_anomalous_days.find().toArray().map(function(doc) {
    // Переводим дату в ключ-строку: "YYYY-MM-DD"
    var y = doc._id.year;
    var m = doc._id.month.toString().padStart(2, "0");
    var d = doc._id.day.toString().padStart(2, "0");
    return y + "-" + m + "-" + d;
});

print("Аномальные дни:", anomalousDates);

// 1. MAP
var map_ips_anomalous_days = function() {
    var dt = new Date(this.timestamp);
    var y = dt.getFullYear();
    var m = (dt.getMonth() + 1).toString().padStart(2, "0");
    var d = dt.getDate().toString().padStart(2, "0");

    var key_date = y + "-" + m + "-" + d;

    if (anomalous.indexOf(key_date) !== -1) {
        emit(this.source_ip, this.bytes_transferred);
    }
};

// 2. REDUCE
var reduce_ips_anomalous_days = function(key, values) {
    return Array.sum(values);
};

// 3. ЗАПУСК MR
db.logs.mapReduce(
    map_ips_anomalous_days,
    reduce_ips_anomalous_days,
    {
        out: "mr_ips_in_anomalous_days",
        scope: { anomalous: anomalousDates }
    }
);

print("MR-3 завершён: mr_ips_in_anomalous_days создана");

```
