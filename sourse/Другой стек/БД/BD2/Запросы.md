```
db.logs.aggregate([
  {
    $group: {
      _id: {
        year: { $year: "$timestamp" },
        month: { $month: "$timestamp" },
        day: { $dayOfMonth: "$timestamp" }
      },
      total_bytes: { $sum: "$bytes_transferred" }
    }
  },
  { $sort: { "_id.year": 1, "_id.month": 1, "_id.day": 1 } }
])
```

```
db.logs.aggregate([
  {
    $group: {
      _id: "$source_ip",
      total_bytes: { $sum: "$bytes_transferred" }
    }
  },
  { $sort: { total_bytes: -1 } },
  { $limit: 5 }
])
```


```
db.logs.aggregate([
  { $match: { action: "blocked" } },
  {
    $group: {
      _id: "$log_type",
      blocked_count: { $sum: 1 }
    }
  }
])
```

db.logs.aggregate([

    }
  },

])
