Нам нужно связать timestamp с нужной датой и часом:

```
INSERT INTO dim_time(timestamp, date_id, hour_id)
SELECT 
    r.timestamp,
    d.date_id,
    h.hour_id 
FROM raw_logs r 
JOIN dim_date d SELECT
  ON d.day = EXTRACT(DAY FROM r.timestamp)::INT
 AND d.month = EXTRACT(MONTH FROM r.timestamp)::INT
 AND d.year = EXTRACT(YEAR FROM r.timestamp)::INT
JOIN dim_hour h 
  ON h.hour = EXTRACT(HOUR FROM r.timestamp)::INT;

```

## **факт-таблицу**

```sql
CREATE TABLE fact_network_events (
    event_id SERIAL PRIMARY KEY,
    timestamp_id INT REFERENCES dim_time(timestamp_id),
    source_ip_id INT REFERENCES dim_source_ip(source_ip_id),
    dest_ip_id INT REFERENCES dim_dest_ip(dest_ip_id),
    protocol_id INT REFERENCES dim_protocol(protocol_id),
    action_id INT REFERENCES dim_action(action_id),
    threat_label_id INT REFERENCES dim_threat_label(threat_label_id),
    log_type_id INT REFERENCES dim_log_type(log_type_id),
    bytes_transferred BIGINT
);

```

- Сначала для каждой строки `raw_logs` ищем соответствующие ID из таблиц измерений.
    
- Потом вставляем факты + ссылки на измерения.
    
```sql
INSERT INTO fact_network_events(
    timestamp_id, source_ip_id, dest_ip_id, protocol_id,
    action_id, threat_label_id, log_type_id, bytes_transferred
)
SELECT 
    t.timestamp_id,
    s.source_ip_id,
    d.dest_ip_id,
    p.protocol_id,
    a.action_id,
    th.threat_label_id,
    l.log_type_id,
    r.bytes_transferred 
FROM raw_logs r 
JOIN dim_time t ON t.timestamp = r.timestamp 
JOIN dim_source_ip s ON s.source_ip = r.source_ip 
JOIN dim_dest_ip d ON d.dest_ip = r.dest_ip 
JOIN dim_protocol p ON p.protocol = r.protocol 
JOIN dim_action a ON a.action = r.action 
JOIN dim_threat_label th ON th.threat_label = r.threat_label 
JOIN dim_log_type l ON l.log_type = r.log_type; 

```


```SQL
INSERT INTO fact_network_events(
    timestamp_id, 
    source_ip_id, 
    dest_ip_id, 
    protocol_id, 
    action_id, 
    threat_label_id, 
    log_type_id, 
    bytes_transferred
) 
SELECT 
    t.timestamp_id,
    s.source_ip_id,
    d.dest_ip_id,
    p.protocol_id,
    a.action_id,
    th.threat_label_id,
    l.log_type_id,
    r.bytes_transferred 
FROM raw_logs r 
JOIN dim_time t ON t.timestamp = r.timestamp 
JOIN dim_source_ip s  ON s.source_ip = r.source_ip 
JOIN dim_dest_ip d  ON d.dest_ip = r.dest_ip 
JOIN dim_protocol p  ON p.protocol = r.protocol 
JOIN dim_action a  ON a.action = r.action 
JOIN dim_threat_label th  ON th.threat_label = r.threat_label 
JOIN dim_log_type l  ON l.log_type = r.log_type 
LIMIT 500000 OFFSET 0;
```
